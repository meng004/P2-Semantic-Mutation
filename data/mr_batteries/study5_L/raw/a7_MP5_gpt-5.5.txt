```python
import math

def _finite_pair(a, b):
    return math.isfinite(float(a)) and math.isfinite(float(b))

# "Refined" regime: move x halfway toward the upper end of the valid interval.
def r_1(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.5
    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    return 0.5 * (x + 1.0)

def R_1(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    return float(y_new) >= float(y_orig)


# Alternative refinement map with a different rate: quadratic complement.
def r_2(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.75
    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    return 1.0 - (1.0 - x) * (1.0 - x)

def R_2(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    return float(y_new) >= float(y_orig)


# Stronger refinement: fixed contraction of the remaining distance to x = 1.
def r_3(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.875
    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    return 1.0 - 0.25 * (1.0 - x)

def R_3(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    return float(y_new) >= float(y_orig)
```
