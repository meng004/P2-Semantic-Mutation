```python
import math

# Each transformation moves the input to a comparable "refined" regime closer
# to the upper end of the same scalar problem domain.  For the correct kernel,
# the solved root is strictly increasing in x, so the refined evaluation should
# not be smaller than the original one.


def r_1(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.0
    x = min(1.0, max(0.0, x))
    return 0.5 + 0.5 * x


def R_1(y_orig, y_new):
    try:
        y_orig = float(y_orig)
        y_new = float(y_new)
    except Exception:
        return False
    return math.isfinite(y_orig) and math.isfinite(y_new) and (y_new >= y_orig)


def r_2(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.0
    x = min(1.0, max(0.0, x))
    return math.sqrt(x)


def R_2(y_orig, y_new):
    try:
        y_orig = float(y_orig)
        y_new = float(y_new)
    except Exception:
        return False
    return math.isfinite(y_orig) and math.isfinite(y_new) and (y_new >= y_orig)


def r_3(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.0
    x = min(1.0, max(0.0, x))
    return 1.0 - (1.0 - x) * (1.0 - x)


def R_3(y_orig, y_new):
    try:
        y_orig = float(y_orig)
        y_new = float(y_new)
    except Exception:
        return False
    return math.isfinite(y_orig) and math.isfinite(y_new) and (y_new >= y_orig)
```
