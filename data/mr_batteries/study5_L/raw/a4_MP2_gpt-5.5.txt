import math

_TOL = 1e-12


def _as_finite_pair(y_orig, y_new):
    try:
        yo = float(y_orig)
        yn = float(y_new)
    except (TypeError, ValueError, OverflowError):
        return None
    if not (math.isfinite(yo) and math.isfinite(yn)):
        return None
    return yo, yn


# Move x upward halfway toward the top element 1.
def r_1(x):
    x = float(x)
    return 0.5 * (x + 1.0)


def R_1(y_orig, y_new):
    pair = _as_finite_pair(y_orig, y_new)
    if pair is None:
        return False
    yo, yn = pair
    return yn + _TOL >= yo


# sqrt(x) is an order-increasing transformation on [0, 1]: sqrt(x) >= x.
def r_2(x):
    x = float(x)
    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    return math.sqrt(x)


def R_2(y_orig, y_new):
    pair = _as_finite_pair(y_orig, y_new)
    if pair is None:
        return False
    yo, yn = pair
    return yn + _TOL >= yo


# Add a positive offset while staying inside [0, 1].
def r_3(x):
    x = float(x)
    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    return min(1.0, x + 0.125)


def R_3(y_orig, y_new):
    pair = _as_finite_pair(y_orig, y_new)
    if pair is None:
        return False
    yo, yn = pair
    return yn + _TOL >= yo
