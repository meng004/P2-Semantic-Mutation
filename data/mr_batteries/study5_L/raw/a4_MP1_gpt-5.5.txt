import math

_CONSERVED_SUM = 8.0 / 3.0
_TOL = 1e-12


def r_1(x):
    return 1.0 - float(x)


def R_1(y_orig, y_new):
    try:
        y0 = float(y_orig)
        y1 = float(y_new)
    except Exception:
        return False
    if not (math.isfinite(y0) and math.isfinite(y1)):
        return False
    return abs((y0 + y1) - _CONSERVED_SUM) <= _TOL
