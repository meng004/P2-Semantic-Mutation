```python
import math

_TOL = 1e-12

def _effective_N(x):
    h = max(float(x), 1e-4)
    return max(4, round(1.0 / h))

def _same_output(y_orig, y_new):
    try:
        yo = float(y_orig)
        yn = float(y_new)
    except Exception:
        return False
    if not (math.isfinite(yo) and math.isfinite(yn)):
        return False
    return abs(yo - yn) <= _TOL * max(1.0, abs(yo), abs(yn))

# Conservation/invariance MR: all inputs inducing the same effective grid N
# conserve the returned convergence ratio exactly up to roundoff.

def r_1(x):
    N = _effective_N(x)
    return 1.0 / float(N)

def R_1(y_orig, y_new):
    return _same_output(y_orig, y_new)

def r_2(x):
    N = _effective_N(x)
    return 1.0 / (float(N) + 0.25)

def R_2(y_orig, y_new):
    return _same_output(y_orig, y_new)

def r_3(x):
    N = _effective_N(x)
    return 1.0 / (float(N) - 0.25)

def R_3(y_orig, y_new):
    return _same_output(y_orig, y_new)
```
