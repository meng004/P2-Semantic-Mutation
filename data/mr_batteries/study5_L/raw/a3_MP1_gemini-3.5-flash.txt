```python
import math
import numpy as np

# MP1 — conservation (invariance under a group action G on the input space)
#
# The program(x) computes the convergence ratio of an explicit Finite Difference Method (FDM)
# for the 1D heat equation.
# Inside the program, the grid spacing h is discretized to:
#   N = max(4, round(1.0 / h))
#   h_act = 1.0 / N
#
# This discretization maps any continuous value x (where h = max(x, 1e-4)) to an integer N.
# Therefore, any two input values x1 and x2 that map to the exact same integer N will lead to
# the exact same numerical grid (h_act), the same time step size (dt), and thus the exact same
# numerical solution and output ratio.
#
# We construct transformations r_k(x) that alter the input x but guarantee that the resulting
# N = round(1.0 / r_k(x)) is identical to N_orig = round(1.0 / x), preserving the output exactly.


def _get_N(x: float) -> int:
    """Helper to compute the discretized number of intervals N for a given input x."""
    h = max(float(x), 1e-4)
    return max(4, round(1.0 / h))


# --- MR1: Minor perturbation within the same discretization bin ---
# For any x, we can perturb x slightly to x_new such that round(1/x_new) == round(1/x).
# This is safe because round(1/x) is a step function.
def r_1(x: float) -> float:
    x = max(float(x), 1e-4)
    N = _get_N(x)
    # Target the exact center of the bin that maps to N
    h_target = 1.0 / N
    # Clamping to [0, 1] range safely
    return min(max(h_target, 1e-4), 1.0)


def R_1(y_orig: float, y_new: float) -> bool:
    if not np.isfinite(y_orig) or not np.isfinite(y_new):
        return False
    # Since the discretization is identical, the outputs must be exactly equal.
    return y_orig == y_new


# --- MR2: Perturbation towards the upper boundary of the same bin ---
# We map x to a value slightly larger than the bin center, but still strictly
# within the same rounding bin, keeping N identical.
def r_2(x: float) -> float:
    x = max(float(x), 1e-4)
    N = _get_N(x)
    # The bin for N is defined by round(1/h) == N, which means:
    # N - 0.5 <= 1/h < N + 0.5  =>  1/(N + 0.5) < h <= 1/(N - 0.5)
    # We choose a point 25% of the way from the center (1/N) to the upper bound 1/(N - 0.5).
    h_center = 1.0 / N
    if N > 4:
        h_upper = 1.0 / (N - 0.49)
    else:
        h_upper = 1.0 / 3.51  # For N=4, the lower bound of N is handled by max(4, ...)
    h_new = h_center + 0.25 * (h_upper - h_center)
    return min(max(h_new, 1e-4), 1.0)


def R_2(y_orig: float, y_new: float) -> bool:
    if not np.isfinite(y_orig) or not np.isfinite(y_new):
        return False
    return y_orig == y_new


# --- MR3: Perturbation towards the lower boundary of the same bin ---
# We map x to a value slightly smaller than the bin center, but still strictly
# within the same rounding bin, keeping N identical.
def r_3(x: float) -> float:
    x = max(float(x), 1e-4)
    N = _get_N(x)
    h_center = 1.0 / N
    h_lower = 1.0 / (N + 0.49)
    h_new = h_center - 0.25 * (h_center - h_lower)
    return min(max(h_new, 1e-4), 1.0)


def R_3(y_orig: float, y_new: float) -> bool:
    if not np.isfinite(y_orig) or not np.isfinite(y_new):
        return False
    return y_orig == y_new
```
