import math

_TOL = 1e-12


def _finite_pair(a, b):
    return math.isfinite(a) and math.isfinite(b)


def _close(a, b, tol=_TOL):
    return abs(a - b) <= tol * max(1.0, abs(a), abs(b))


# MR1: compare original x with a refined/shifted regime closer to x=1:
# r = (x + 1) / 2.  For the correct kernel,
# y_new = 0.5*y_orig + 7/6 and y_new >= y_orig.
def r_1(x):
    x = float(x)
    return 0.5 * (x + 1.0)


def R_1(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    expected = 0.5 * y_orig + 7.0 / 6.0
    return y_new + _TOL >= y_orig and _close(y_new, expected)


# MR2: a milder refinement toward x=1:
# r = (3x + 1) / 4.  For the correct kernel,
# y_new = 0.75*y_orig + 7/12 and y_new >= y_orig.
def r_2(x):
    x = float(x)
    return 0.75 * x + 0.25


def R_2(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    expected = 0.75 * y_orig + 7.0 / 12.0
    return y_new + _TOL >= y_orig and _close(y_new, expected)


# MR3: nonlinear refinement r = sqrt(x), which remains in [0,1] and satisfies
# r >= x.  Since y = 2x + 1/3, infer x from y_orig and compare relatively.
def r_3(x):
    x = float(x)
    if x <= 0.0:
        return 0.0
    if x >= 1.0:
        return 1.0
    return math.sqrt(x)


def R_3(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    x_est = 0.5 * (y_orig - 1.0 / 3.0)
    if x_est < -_TOL or x_est > 1.0 + _TOL:
        return False
    x_est = min(1.0, max(0.0, x_est))
    expected = 2.0 * math.sqrt(x_est) + 1.0 / 3.0
    return y_new + _TOL >= y_orig and _close(y_new, expected)
