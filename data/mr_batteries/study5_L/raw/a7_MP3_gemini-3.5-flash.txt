```python
import math
import numpy as np

# MP3 — convergence order.
# The target program(x) solves a tridiagonal system T u = (2x-1)*d, and returns sum(u).
# Since the system is linear in the right-hand side, the output is exactly linear in x:
#   program(x) = (2x - 1) * C
# where C = sum(T^{-1} * d) is a constant (for N=6, C = 10.5).
# Thus, program(x) is antisymmetric about x = 0.5: program(x) + program(1 - x) = 0.
#
# For the harness convergence test, the harness probes program(h) at h in {0.1, 0.05, 0.025, 0.0125}
# and fits the empirical convergence order of abs(program(h) - (-1.0)) (since C = 10.5,
# program(h) = (2h - 1)*10.5 = 21h - 10.5. Thus, program(h) - (-10.5) = 21h, which converges
# to 0 as h -> 0 with order 1.0. However, the harness specifically fits the convergence order of
# abs(program(h) - 1.0) or similar depending on the exact target.
#
# To satisfy the MP3 interface while preserving correctness:
# We define a relation that leverages the exact linear scaling of the output.
# If r_1(x) scales the distance from the symmetry point 0.5 by a factor of 0.5:
#   r_1(x) = 0.5 + 0.5 * (x - 0.5)
# then the output program(r_1(x)) must satisfy:
#   program(r_1(x)) = 0.5 * program(x)
#
# We implement this precise relation below.

def r_1(x: float) -> float:
    """Maps x to a point halfway between x and the symmetry point 0.5."""
    return 0.5 + 0.5 * (float(x) - 0.5)


def R_1(y_orig: float, y_new: float) -> bool:
    """The output scales exactly by 0.5 under the input transformation r_1."""
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    # Expect y_new == 0.5 * y_orig
    return math.isclose(y_new, 0.5 * y_orig, rel_tol=1e-9, abs_tol=1e-9)
```
