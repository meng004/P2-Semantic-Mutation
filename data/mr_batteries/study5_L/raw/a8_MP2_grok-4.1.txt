import math
import numpy as np

T_END = 1.0
N_STEPS = 20

def _rhs(u):
    return -u

def program(x) -> float:
    u = 2.0 * float(x) - 1.0
    h = T_END / N_STEPS
    for _ in range(N_STEPS):
        k1 = _rhs(u)
        k2 = _rhs(u + 0.5 * h * k1)
        k3 = _rhs(u + 0.5 * h * k2)
        k4 = _rhs(u + h * k3)
        u = u + h / 6.0 * (k1 + 2.0 * k2 + 2.0 * k3 + k4)
    return float(u)

# MP2 MR family: monotonicity (linear increasing order on x in [0,1])
# r_1: identity (trivial case)
def r_1(x):
    return float(x)

def R_1(y_orig, y_new):
    return y_new >= y_orig

# r_2: exponential (strictly increasing, concave)
def r_2(x):
    return math.exp(x)

def R_2(y_orig, y_new):
    return y_new >= y_orig

# r_3: hyperbolic tangent (strictly increasing, concave)
def r_3(x):
    return np.tanh(x)

def R_3(y_orig, y_new):
    return y_new >= y_orig
