```python
import math

# Move the evaluation point toward the symmetry/maximum point x=0.5.
# For the natural cubic spline of sin(pi*x), this should not decrease the value.

def r_1(x):
    x = float(x)
    return 0.5 + 0.5 * (x - 0.5)

def R_1(y_orig, y_new):
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    return y_new >= y_orig - 1e-14
```
