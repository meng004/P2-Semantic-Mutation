import math

_ABS_TOL = 1e-6
_REL_TOL = 1e-12


def r_1(x):
    # Identity transformation: the only exact shape/output-preserving
    # transformation exposed by this scalar input interface.
    return float(x)


def R_1(y_orig, y_new):
    try:
        a = float(y_orig)
        b = float(y_new)
    except Exception:
        return False
    if not (math.isfinite(a) and math.isfinite(b)):
        return False
    return math.isclose(a, b, rel_tol=_REL_TOL, abs_tol=_ABS_TOL)
