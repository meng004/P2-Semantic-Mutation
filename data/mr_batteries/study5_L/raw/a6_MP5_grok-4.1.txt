def r_1(x):
    """
    MR1: Squared-input regime (near 0).
    Rationale (MP5-MODE-M): Input transformation x -> x**2 maps [0,1] -> [0,1] to a comparable regime (smaller x yields smaller |r|). Output relation asserts |r_new| <= |r_orig| (higher fidelity/accuracy for smaller root).
    """
    return x**2


def R_1(y_orig, y_new):
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
