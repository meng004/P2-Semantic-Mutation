import numpy as np

    """Map x to [0, 0.4] and shift it right by 0.1.

    Since the spline is strictly increasing on [0, 0.5],
    S(x_new) >= S(x_orig) must hold.

    x_orig = (x % 1.0) * 0.4
    # Shift right by 0.1, ensuring the new point is in [0.1, 0.5]
    x_new = x_orig + 0.1

    # To pass x_orig implicitly, we can encode it in the float, but since the

    # x_new = 0.4 * x + 0.1
