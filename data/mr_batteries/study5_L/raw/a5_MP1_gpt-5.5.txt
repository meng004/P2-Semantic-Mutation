import math

_TOL = 1e-12

def r_1(x):
    # Reflection symmetry of sin(pi*t) and the natural spline on symmetric knots.
    return 1.0 - float(x)

def R_1(y_orig, y_new):
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    return math.isclose(float(y_orig), float(y_new), rel_tol=_TOL, abs_tol=_TOL)
