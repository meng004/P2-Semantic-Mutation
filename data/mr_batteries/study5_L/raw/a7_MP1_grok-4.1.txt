import math
import numpy as np

def r_1(x):
    """r_1: x -> 1-x (direct symmetry)"""
    return float(1.0 - x)

def R_1(y_orig, y_new):
    """R_1: Σu(x) + Σu(1-x) == 0"""
    return math.isclose(y_orig + y_new, 0.0)

def r_2(x):
    """r_2: x -> 0.5 - (x - 0.5) (antisymmetry)"""
    return float(0.5 - (x - 0.5))

def R_2(y_orig, y_new):
    """R_2: Σu(0.5 - (x - 0.5)) == -Σu(x)"""
    return math.isclose(y_new, -y_orig)

def r_3(x):
    """r_3: x -> 1 - (x - 0.5)**2 (composite linear+quadratic)"""
    t = 1.0 - (x - 0.5)**2
    return float(max(0.0, min(1.0, t)))

def R_3(y_orig, y_new):
    """R_3: Σu(1 - (x-0.5)^2) == -(Σu(x) + Σu(1-x))"""
    return math.isclose(y_new, -(y_orig + y_orig))  # since y_orig == -y for r1 on same pair
