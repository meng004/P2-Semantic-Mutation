import numpy as np

def _g(r, rhs):
    return r**3 + r - rhs

def program(x):
    x = float(x)
    rhs = 4.0 * x - 2.0
    return float(brentq(_g, -5.0, 5.0, args=(rhs,), xtol=1e-12, rtol=8.9e-16))

# Module-level constants for conservation (r(x) + r(1-x) = 0)

EPSILON = 1e-12

# MR1: Direct odd symmetry conservation (parent m_inv — transformation G: x -> 1-x)
def r_1(x):
    return 1.0 - x

def R_1(y_orig, y_new):
    return abs(y_orig + y_new) <= (EPSILON * (abs(y_orig) + abs(y_new)) + 1e-15)

# MR2: Reflection symmetry (parent m_inv — transformation G: x -> 0.5 + 0.5 * cos(2 * pi * x))

    return 0.5 + 0.5 * np.cos(2.0 * math.pi * x)

def R_2(y_orig, y_new):
    return abs(y_orig + y_new) <= (EPSILON * (abs(y_orig) + abs(y_new)) + 1e-15)
