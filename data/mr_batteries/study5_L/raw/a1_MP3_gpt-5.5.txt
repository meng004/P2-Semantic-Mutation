import math

def r_1(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.0
    return min(1.0, max(0.0, x))

def R_1(y_orig, y_new):
    return math.isfinite(float(y_orig)) and math.isfinite(float(y_new))
