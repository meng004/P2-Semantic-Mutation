```python
import math

_EPS = 1e-12


def _finite_geq(y_orig, y_new):
    try:
        yo = float(y_orig)
        yn = float(y_new)
    except Exception:
        return False
    if not (math.isfinite(yo) and math.isfinite(yn)):
        return False
    return yn + _EPS >= yo


def r_1(x):
    x = float(x)
    return 0.5 * (x + 1.0)


def R_1(y_orig, y_new):
    return _finite_geq(y_orig, y_new)


def r_2(x):
    x = float(x)
    return math.sqrt(x)


def R_2(y_orig, y_new):
    return _finite_geq(y_orig, y_new)


def r_3(x):
    x = float(x)
    return min(1.0, x + 0.25)


def R_3(y_orig, y_new):
    return _finite_geq(y_orig, y_new)
```
