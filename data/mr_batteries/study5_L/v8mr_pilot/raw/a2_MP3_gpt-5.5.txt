import math

def r_1(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.0
    return min(1.0, max(0.0, x / 2.0))

def R_1(y_orig, y_new):
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    return abs(y_new - 1.0) <= abs(y_orig - 1.0) + 1e-12
