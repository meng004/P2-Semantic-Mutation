```python
import math
import numpy as np


# MR1: Conservation of determinant under the complement transform x -> 1 - x.
# By definition of the program, det(A(x)) + det(A(1 - x)) = 15.
# Thus, the sum of the outputs for x and 1 - x must be exactly 15 (within numerical tolerance).
def r_1(x: float) -> float:
    return 1.0 - float(x)


def R_1(y_orig: float, y_new: float) -> bool:
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    # Check if the sum is close to 15.0
    return math.isclose(y_orig + y_new, 15.0, rel_tol=1e-9, abs_tol=1e-9)
```
