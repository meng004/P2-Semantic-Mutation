import math

_TOL = 1e-9
_CONSERVED_SUM = 15.0

def r_1(x):
    x = float(x)
    return 1.0 - x

def R_1(y_orig, y_new):
    try:
        y_orig = float(y_orig)
        y_new = float(y_new)
    except Exception:
        return False
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    return math.isclose(y_orig + y_new, _CONSERVED_SUM, rel_tol=_TOL, abs_tol=_TOL)
