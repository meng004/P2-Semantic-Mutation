import math

_TOL = 1e-12


def r_1(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.0
    return min(1.0, max(0.0, x))


def R_1(y_orig, y_new):
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    return abs(y_new - y_orig) <= _TOL


def r_2(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.0
    # Trivial identity action, written as the inverse of itself on [0, 1].
    z = 1.0 - (1.0 - x)
    return min(1.0, max(0.0, z))


def R_2(y_orig, y_new):
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    return abs(y_new - y_orig) <= 1e-11


def r_3(x):
    x = float(x)
    if not math.isfinite(x):
        return 0.0
    # Identity element of the conservation/invariance group action.
    z = math.fsum((x, 0.0))
    return min(1.0, max(0.0, z))


def R_3(y_orig, y_new):
    if not (math.isfinite(y_orig) and math.isfinite(y_new)):
        return False
    return math.isclose(y_orig, y_new, rel_tol=1e-12, abs_tol=1e-12)
