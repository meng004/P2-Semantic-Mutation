```python
import math

_EPS = 1e-12


def _finite_pair(a, b):
    return math.isfinite(float(a)) and math.isfinite(float(b))


def r_1(x):
    x = float(x)
    return 0.5 * (x + 1.0)


def R_1(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    return float(y_new) >= float(y_orig) - _EPS


def r_2(x):
    x = float(x)
    return math.sqrt(max(0.0, min(1.0, x)))


def R_2(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    return float(y_new) >= float(y_orig) - _EPS


def r_3(x):
    x = float(x)
    return min(1.0, max(0.0, x) + 0.25)


def R_3(y_orig, y_new):
    if not _finite_pair(y_orig, y_new):
        return False
    return float(y_new) >= float(y_orig) - _EPS
```
